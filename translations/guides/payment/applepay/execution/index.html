执行支付处理

在 fincode 中，Apple Pay 支付从支持 Apple Pay 的 iPhone、iPad、Apple Watch 或 Mac 获取支付令牌开始。
向 Apple 获取支付令牌的详细规格，请参阅 Apple Developer 的 PassKit（Apple Pay and Wallet）中的 Offering Apple Pay in Your App。

A. 在网站中获取支付令牌（JavaScript）

在网站中处理 Apple Pay 支付时，可以使用受支持设备浏览器提供的 Apple Pay JS 或 Payment Request API。
本页示例使用 Apple Pay JS。

※有关自定义付款按钮样式的方法，请参阅 Apple Developer 的 Displaying Apple Pay Buttons Using JavaScript。
※详细规格请参阅 Apple Developer 的 Apple Pay on the Web。

启动 Apple Pay 付款会话（example.js）

const APPLE_PAY_SUPPORTED_VERSION = 3; // Apple PayJSのサポートバージョン
const MARCHANT_IDENTIFIER = "example.com.store"; // マーチャントID
const APPLE_PAY_BUTTON_ID = "apple-pay-button"; // 任意のApple Payボタン要素のID名
const SHOP_VALIDATION_ENDPOINT_URL =
    "https://your-server.example.com/applepay-session/validate"; // 任意のマーチャントセッション取得エンドポイントのURL
     
/* ブラウザがApple PayJSをサポートしている場合 */
if (window.ApplePaySession) {
    const promise =
        ApplePaySession.canMakePaymentsWithActiveCard(MARCHANT_IDENTIFIER);
    promise.then((canMakePayment) => {
        /* ユーザーが少なくとも1枚の利用可能なカードを登録している場合 */
        if (canMakePayment) {
            // Apple Payボタンを作成
            const applePayButton = document.createElement("button");
            applePayButton.setAttribute("id", APPLE_PAY_BUTTON_ID);
            applePayButton.setAttribute("lang", "ja-JP");
            const applePayButtonType = "buy"; // Apple Payボタンの種類, styleプロパティ -apple-pay-button-type の値
            const applePayButtonStyle = "black"; // Apple Payボタンのスタイル, styleプロパティ -apple-pay-button-style の値
            applePayButton.setAttribute(
                "style",
                `-webkit-appearance: -apple-pay-button; -apple-pay-button-type: ${applePayButtonType}; -apple-pay-button-style: ${applePayButtonStyle};`
            );
     
            // Apple PayボタンをDOMに追加
            const applePayButtonContainer = document.getElementById(
                "{Apple Payボタンを追加したい要素のID}"
            );
            applePayButtonContainer.appendChild(applePayButton);
     
            // Apple Payボタンをクリックしたときのイベントハンドラ
            applePayButton.addEventListener("click", (_) => {
                // Apple Pay決済リクエスト
                // 参照: https://developer.apple.com/documentation/apple_pay_on_the_web/applepaypaymentrequest
                const applePayPaymentRequest = {
                    countryCode: "JP",
                    currencyCode: "JPY",
                    supportedNetworks: ["visa", "masterCard", "jcb", "amex"],
                    merchantCapabilities: ["supports3DS"],
                    total: {
                        label: "{商品名}",
                        amount: "{金額}",
                    },
                };
     
                // Apple Payセッションを作成
                // 参照: https://developer.apple.com/documentation/apple_pay_on_the_web/applepaysession/2320659-applepaysession
                const applePaySession = new ApplePaySession(
                    APPLE_PAY_SUPPORTED_VERSION,
                    applePayPaymentRequest
                );
     
                // Apple Payの支払いシートが表示され、マーチャントセッションの取得を求められたときのイベントハンドラ
                // 参照: https://developer.apple.com/documentation/apple_pay_on_the_web/applepaysession/1778021-onvalidatemerchant
                applePaySession.onvalidatemerchant = (event) => {
                    const validationURL = event.validationURL;
     
                    // マーチャントセッションを取得
                    fetch(SHOP_VALIDATION_ENDPOINT_URL, {
                        method: "POST",
                        headers: {
                            "Content-Type": "application/json",
                        },
                        body: JSON.stringify({
                            validationURL: validationURL,
                        }),
                    }).then((response) => {
                        // マーチャントセッションをApple Payに返却
                        applePaySession.completeMerchantValidation(response);
                    });
                };
     
                // Apple Payのキャンセルボタンが押されたときのイベントハンドラ
                // 参照: https://developer.apple.com/documentation/apple_pay_on_the_web/applepaysession/1778029-oncancel
                applePaySession.oncancel = (event) => {};
     
                // Apple Pay上で支払い方法が選択されたときのイベントハンドラ
                // 参照: https://developer.apple.com/documentation/apple_pay_on_the_web/applepaysession/1778013-onpaymentmethodselected
                applePaySession.onpaymentmethodselected = (event) => {};
     
                // Apple Pay上で配送方法が選択されたときのイベントハンドラ
                // 参照: https://developer.apple.com/documentation/apple_pay_on_the_web/applepaysession/1778028-onshippingmethodselected
                applePaySession.onshippingmethodselected = (event) => {};
     
                // Apple Pay上で配送先住所が選択されたときのイベントハンドラ
                // 参照: https://developer.apple.com/documentation/apple_pay_on_the_web/applepaysession/1778009-onshippingcontactselected
                applePaySession.onshippingcontactselected = (event) => {};
     
                // 購入者により購入が承認されたときのイベントハンドラ
                // 参照: https://developer.apple.com/documentation/apple_pay_on_the_web/applepaysession/1778020-onpaymentauthorized
                applePaySession.onpaymentauthorized = (event) => {
                    const token = event.payment.token;
     
                    // トークンをBase64エンコード
                    // このBase64エンコード済みトークンを決済実行JSの引数もしくは決済実行APIのリクエストボディに含めます
                    const encodedToken = btoa(JSON.stringify(token.paymentData));
     
                    /**
                     * 後続処理
                     */
                };
     
                // Apple Payセッションを開始
                applePaySession.begin();
            });
        }
    });
}

B. 在 iOS、iPadOS、watchOS 或 macOS 中获取支付令牌（Swift）

在 Apple 设备上的应用中处理 Apple Pay 支付时，可以使用 Apple 提供的 PassKit。

※详细规格请参阅 Apple Developer 的 PassKit（Apple Pay and Wallet）中的 Offering Apple Pay in Your App。

启动 Apple Pay 付款会话（sample.swift）

import UIKit
import PassKit
     
let APPLE_PAY_MERCHANT_ID = "example.com.store"
     
class ViewController: UIViewController, PKPaymentAuthorizationViewControllerDelegate {
     
    // Apple Payボタン
    @IBOutlet weak var applePayButton: UIView!
     
    override func viewDidLoad() {
        super.viewDidLoad()
     
        let isApplePayAvailable = isApplePayAvailable()
     
        /* もしApple Payが利用できる場合は、Apple Payボタンを表示する */
        if isApplePayAvailable {
     
            // Apple Payボタンを作成
            let button = PKPaymentButton(type: .buy, style: .black)
            button.addAction(
                UIAction(handler: { _ in
                    self.onApplePayButtonTapped()
                }),
                for: .touchUpInside
            )
     
            // Apple Payボタンを表示
            applePayButton.addSubview(button)
        }
    }
     
    // Apple Payで利用できるカードが少なくとも1つあるかどうかを判定
    func isApplePayAvailable() -> Bool {
        if !PKPaymentAuthorizationViewController.canMakePayments() {
            // Apple Payは利用できない
            return false
        }
     
        let networks = PKPaymentRequest.availableNetworks()
        if !PKPaymentAuthorizationViewController.canMakePaymentsUsingNetworks(networks) {
            // Apple Pay機能は利用できるが、利用できるカードが登録されていない
            return false
        }
     
        return true
    }
     
    // Apple Payボタンがタップされた時の処理
    func onApplePayButtonTapped() {
        // Apple Payで支払いを行う
        let request = PKPaymentRequest()
        request.merchantIdentifier = APPLE_PAY_MERCHANT_ID
        request.supportedNetworks = [.visa, .masterCard, .amex, .JCB]
        request.merchantCapabilities = .capability3DS
        request.countryCode = "JP"
        request.currencyCode = "JPY"
        request.paymentSummaryItems = [
            PKPaymentSummaryItem(
                label: "{{商品名}}",
                amount: NSDecimalNumber(string: "{{金額}}")
            )
        ]
     
        let viewController = PKPaymentAuthorizationViewController(paymentRequest: request)
        viewController.delegate = self
        self.present(viewController, animated: true, completion: nil)
    }
     
    // Apple Payで支払いが完了した時の処理
    func paymentAuthorizationViewControllerDidFinish(_ controller: PKPaymentAuthorizationViewController) {
        // Apple Payの支払シートを閉じる
        controller.dismiss(animated: true, completion: nil)
    }
     
    // Apple Payで支払いが承認された時の処理
    func paymentAuthorizationViewController(
        _ controller: PKPaymentAuthorizationViewController,
        didAuthorizePayment payment: PKPayment,
        handler completion: @escaping (PKPaymentAuthorizationResult) -> Void
    ) {
        let token = payment.token.paymentData
     
        // トークンをBase64エンコード
        // このBase64エンコード済みのトークンを決済実行APIのリクエストに含めます
        let encodedToken = token.base64EncodedString()
     
        /*
            後続処理
        */
     
        // Apple Payの支払シートを閉じる
        controller.dismiss(animated: true, completion: nil)
     
        // エラーなくApple Payでの支払いが完了したことを通知
        completion(PKPaymentAuthorizationResult(status: .success, errors: nil))
    }
}

使用支付执行 API

在网站或应用中获取令牌后，将令牌填入支付执行 API 请求体的 token 字段。

1. 买家点击“Apple Pay”按钮。
2. 商户调用 fincode 的登记支付接口。
3. 商户从设备获取支付令牌。
4. 商户将支付令牌填入 fincode 的トークン(token)，并执行支付。
5. 使用令牌处理支付。

使用支付执行 JS

在网站中获取令牌后，将令牌填入支付执行 JS 的 transaction.token 参数。

1. 买家点击“Apple Pay”按钮。
2. 商户调用 fincode 的登记支付接口。
3. 商户从设备获取支付令牌。
4. 商户将支付令牌填入 fincode 的トークン(token)，并调用支付执行 JS。
5. 使用令牌处理支付。
