进行信用卡支付

下面实际执行一次信用卡支付。

使用支付登记 API 和支付执行 API 完成支付。
首先使用支付登记 API 登记支付。请参照下面的示例代码尝试登记。

支付登记 API

终端
curl https://api.test.fincode.jp/v1/payments \
  -H "Content-Type: application/json" \
  -H "Authorization: Bearer m_test_OGY1YWE" \
  -d '{"pay_type": "Card", "job_code": "CAPTURE", "amount": "500"}'
Python
import json
import urllib.request
import urllib.parse

API_KEY = 'Bearer <用户的API密钥>'
BASE_URL = 'https://api.test.fincode.jp'
CHARSET = 'utf-8'
METHOD = 'POST'
DATA = {
    'pay_type': 'Card',
    'job_code': 'CAPTURE',
    'amount': '500',
}

def create_request(endpoint):
    url = BASE_URL + endpoint
    req = urllib.request.Request(url, method=METHOD)
    req.add_header('Content-Type', 'application/json')
    req.add_header('Authorization', API_KEY)
    return req

def main():
    endpoint = '/v1/payments'

    req = create_request(endpoint)

    try:
        with urllib.request.urlopen(req, data=json.dumps(DATA).encode(CHARSET)
                                    ) as res:
            body = res.read().decode(CHARSET)
            print('SUCCESS')
            print(body)
    except urllib.error.HTTPError as err:
        body = err.read().decode(CHARSET)
        print('ERROR')
        print(body)

if __name__ == "__main__":
    main()
Go
package main

import (
    "bytes"
    "fmt"
    "io"
    "log"
    "net/http"
)

const apiKey = "Bearer <用户的API密钥>"
const baseURL = "https://api.test.fincode.jp"
const method = "POST"
const id = "o_test_id"
const data = `{
                "pay_type": "Card",
                "job_code": "CAPTURE",
                "amount":  "500"
            }`

func main() {

    endpoint := "/v1/payments"
    URL := baseURL + endpoint

    req, err := http.NewRequest(
        method,
        URL,
        bytes.NewBuffer([]byte(data)),
    )
    if err != nil {
        log.Fatal(err)
    }

    req.Header.Set("Content-Type", "application/json")
    req.Header.Set("Authorization", apiKey)

    client := &http.Client{}
    resp, err := client.Do(req)
    if err != nil {
        log.Fatal(err)
    }

    bodyJsonBytes, err := io.ReadAll(resp.Body)
    if err != nil {
        log.Fatal(err)
    }
    defer resp.Body.Close()

    if resp.StatusCode != 200 {
        fmt.Println("ERROR")
        fmt.Println(string(bodyJsonBytes))
        return
    }

    fmt.Println("SUCCESS")
    fmt.Println(string(bodyJsonBytes))
}
Node.js
"use strict";

const console = require("console");
const request = require("request");
const API_KEY = "Bearer <用户的API密钥>";
const BASE_URL = "https://api.test.fincode.jp";
const DATA = {
  pay_type: "Card",
  job_code: "CAPTURE",
  amount: "500",
};

main();

function main() {
  var endpoint = "/v1/payments";
  request.post(
    {
      url: BASE_URL + endpoint,
      headers: {
        "Content-Type": "application/json; charset=utf-8",
        Authorization: API_KEY,
      },

      json: DATA,
    },
    (error, res, body) => {
      if (200 != res.statusCode) {
        console.log("ERROR");
        console.log(body);
      } else {
        console.log("SUCCESS");
        console.log(body);
      }
    }
  );
}

出于安全考虑，处理购买者的卡片信息时须将其转换为令牌，或使用支付执行 JS。

本例使用支付执行 JS。该脚本从购买者的浏览器直接调用 fincode 的支付执行 API。卡号不经过商户的服务器。

通过支付执行 JS 完成支付执行后，整个支付流程即告结束。

示例：电商网站（浏览器）中的输入表单

支付执行 JS

JavaScript
function postForm( calltype )
{
  // 读取商户的 API 公钥
  const api_key=<公钥>;
  // 创建 Fincode 实例
  let fincode = Fincode(api_key);
  // 定义执行支付所需的数据
  const transaction = {
    // 订单 ID
    id: document.getElementById("order_id").value,
    // 決済種別
    pay_type: document.getElementById("pay_type").value,
    // 取引ID
    access_id: document.getElementById("access_id").value,
    // 付款方式
    method: document.getElementById("method").value,
    // 卡号
    card_no : document.getElementById("card_no").value,
    // 有効期限 yymm形式
    expire : document.getElementById("expire").value,
    // 持卡人姓名
    holder_name: document.getElementById("holder_name").value,
    // 安全码
    security_code: document.getElementById("security_code").value,
  };

  // 決済実行
  fincode.payments(transaction,
    function (status, response) {
        if (status === 200) {
              // 请求成功时的处理
              console.log(response);
        } else {
                // 请求出错时的处理
                console.log(response);
        }
    },
    function () {
            // 通信错误处理
            console.log("ERROR");
    });
}
